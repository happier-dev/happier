Use External Sessions when an Agent has its own session store.

> **Development/preview contract.** This example describes the current
> source-level authoring and host behavior. It does not claim that loaded or
> packaged release artifacts expose this surface.

## Manifest

Declare the Agent local ID and the `externalSessions` capability right in
`contributes.agents`. The same Agent entry supplies its source descriptor:

```json
{
  "id": "example",
  "title": "Example Agent",
  "capabilities": {
    "surfaces": ["externalSessions"]
  },
  "surfaces": {
    "externalSession": {
      "sources": [{
        "sourceKind": "exampleStore",
        "schema": {
          "fields": [{
            "name": "kind",
            "kind": "literal",
            "value": "exampleStore"
          }]
        },
        "key": {
          "segments": [{
            "kind": "literal",
            "value": "exampleStore"
          }]
        },
        "instances": [{
          "kind": "default",
          "constants": {}
        }]
      }]
    }
  }
}
```

## Authoring

Put the exact six-method contribution on the same code-defined Agent entry.
`definePlugin` projects the manifest and generates its one activation binding:

```ts
import { definePlugin } from '@happier-dev/plugin-sdk';
import type {
  AgentExternalSessionTranscriptRawRecord,
  AgentExternalSessionsContribution,
} from '@happier-dev/plugin-sdk/sessions/external';

// Decode each source row into the canonical transcript record. A
// provider-native envelope fails the whole page with `agent_error`.
const agentTurn: AgentExternalSessionTranscriptRawRecord = {
  role: 'agent',
  content: {
    type: 'acp',
    agentId: 'example',
    data: { type: 'message', message: 'Reviewed the staged diff.' },
  },
};

const externalSessions = {
  resolveSource: ({ source }) => ({ ok: true, value: { source } }),
  listCandidates: () => ({
    ok: true,
    value: { candidates: [], nextCursor: null },
  }),
  resolveLinkIdentity: ({ source, remoteSessionId }) => ({
    ok: true,
    value: { source, remoteSessionId, linkData: {} },
  }),
  resolveLinkedIdentity: ({ source, remoteSessionId, linkData }) => ({
    ok: true,
    value: { source, remoteSessionId, linkData },
  }),
  pageTranscript: () => ({
    ok: true,
    value: {
      items: [{
        id: 'example-1',
        createdAtMs: 1_700_000_000_000,
        messageRole: 'agent',
        raw: agentTurn,
      }],
      nextCursor: null,
    },
  }),
  readAfterTranscript: () => ({
    ok: true,
    value: { outcome: 'already_current' },
  }),
} satisfies AgentExternalSessionsContribution;

export const { manifest, activate } = definePlugin({
  id: 'examples.external-session-source',
  version: '0.1.0',
  entrypoints: { daemon: './dist/index.js' },
  agents: {
    example: {
      declaration: {
        title: 'Example Agent',
        capabilities: { surfaces: ['externalSessions'] },
        surfaces: {
          externalSession: {
            sources: [{
              sourceKind: 'exampleStore',
              schema: {
                fields: [{
                  name: 'kind',
                  kind: 'literal',
                  value: 'exampleStore',
                }],
              },
              key: {
                segments: [{ kind: 'literal', value: 'exampleStore' }],
              },
              instances: [{ kind: 'default', constants: {} }],
            }],
          },
        },
      },
      externalSessions,
    },
  },
});
```
